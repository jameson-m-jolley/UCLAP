basic grammar 


START -> (ARGUMENTS|OPTION|COMMAND) EPSILON

COMMAND -> LITERAL (ARGUMENTS|EPSILON)

OPTION -> FLAG  (ARGUMENTS|EPSILON)

FLAG -> (LONG | SHORT)

ARGUMENT_TYPE -> (POSITIONAL_ARGUMENT|ARBITRARY_ARGUMENT) 

ARGUMENTS -> ARGUMENT_TYPE (ARGUMENTS|EPSILON)

POSITIONAL_ARGUMENT -> (STRING|LITERAL|NUMBER)

ARBITRARY_ARGUMENT -> KEY_VALUE
